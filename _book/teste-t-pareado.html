16  Procedimentos t para Dados Emparelhados

NoteTeste t para Dados Emparelhados

Este capítulo apresenta os métodos estatísticos para análise de dados emparelhados, uma das técnicas mais poderosas em inferência estatística quando temos medições repetidas ou pares relacionados.

16.1 Conceitos Fundamentais

Dados emparelhados ocorrem quando temos duas medições na mesma unidade experimental ou unidades experimentais muito similares. A análise de dados emparelhados é mais poderosa que a comparação de duas amostras independentes porque:

  1. Controla a variabilidade entre unidades
  2. Foca na diferença dentro de cada par
  3. Reduz o erro padrão das estimativas

16.1.1 Quando Usar Dados Emparelhados

TipSituações Típicas para Dados Emparelhados
  1. Estudos antes-depois: Medições na mesma pessoa/unidade em momentos diferentes
  2. Pares naturais: Irmãos gêmeos, cônjuges, vizinhos
  3. Matching experimental: Unidades pareadas por características similares
  4. Medições duplas: Dois métodos aplicados na mesma unidade

16.4 Teste t para Dados Emparelhados

16.4.1 Fundamentos Teóricos

Para dados emparelhados, analisamos as diferenças \(d_i = x_{2i} - x_{1i}\) onde:

  • \(x_{1i}\) e \(x_{2i}\) são as duas medições do par \(i\)
  • As diferenças \(d_i\) seguem uma distribuição Normal
  • Testamos hipóteses sobre \(\mu_d\) (média populacional das diferenças)

Estatística de teste: \[t = \frac{\bar{d} - \mu_{d0}}{s_d/\sqrt{n}}\]

onde \(\bar{d}\) é a média das diferenças amostrais e \(s_d\) é o desvio-padrão das diferenças.

Code
```{r}
#| label: teste-t-emparelhado

# Configurando o teste de hipóteses
# H₀: μ_d = 0 (não há melhora média)
# H₁: μ_d > 0 (há melhora média) - teste unilateral à direita

cat("TESTE T PARA DADOS EMPARELHADOS\n")
cat("===============================\n\n")

cat("Hipóteses:\n")
cat("H₀: μ_d = 0 (não há melhora no desempenho)\n")
cat("H₁: μ_d > 0 (há melhora no desempenho)\n")
cat("α = 0.05\n\n")

# Realizando o teste t pareado
teste_pareado <- t.test(dados_capacitacao$Depois, 
                       dados_capacitacao$Antes,
                       paired = TRUE,
                       alternative = "greater",  # teste unilateral
                       conf.level = 0.95)

# Exibindo resultados detalhados
cat("RESULTADOS DO TESTE:\n")
cat("===================\n")
cat("Estatística t =", round(teste_pareado$statistic, 4), "\n")
cat("Graus de liberdade =", teste_pareado$parameter, "\n")
cat("p-valor =", round(teste_pareado$p.value, 6), "\n")
cat("Intervalo de confiança 95% para μ_d: [", 
    round(teste_pareado$conf.int[1], 3), ", ∞)\n")
cat("Diferença média estimada =", round(teste_pareado$estimate, 3), "\n\n")

# Interpretação do resultado
if(teste_pareado$p.value < 0.05) {
  cat("CONCLUSÃO: Rejeitamos H₀ (p < 0.05)\n")
  cat("Há evidência estatística de melhora no desempenho após o treinamento.\n\n")
} else {
  cat("CONCLUSÃO: Não rejeitamos H₀ (p ≥ 0.05)\n")
  cat("Não há evidência estatística suficiente de melhora no desempenho.\n\n")
}

# Cálculos manuais para verificação
n <- length(diferenças)
media_d <- mean(diferenças)
dp_d <- sd(diferenças)
erro_padrao <- dp_d / sqrt(n)

cat("CÁLCULOS DETALHADOS:\n")
cat("===================\n")
cat("n =", n, "\n")
cat("x̄_d =", round(media_d, 4), "\n")
cat("s_d =", round(dp_d, 4), "\n")
cat("EP(x̄_d) = s_d/√n =", round(erro_padrao, 4), "\n")
cat("t = x̄_d / EP(x̄_d) =", round(media_d/erro_padrao, 4), "\n")

# Valor crítico
valor_critico <- qt(0.95, df = n-1)
cat("Valor crítico t₀.₀₅,₁₁ =", round(valor_critico, 4), "\n")
cat("Decisão:", ifelse(teste_pareado$statistic > valor_critico, 
                      "t > t_crítico → Rejeitar H₀", 
                      "t ≤ t_crítico → Não rejeitar H₀"), "\n")
```

16.5 Intervalo de Confiança para a Diferença Média

Code
```{r}
#| label: ic-diferenca-media

cat("INTERVALO DE CONFIANÇA BILATERAL\n")
cat("================================\n\n")

# IC bilateral de 95% para μ_d
ic_bilateral <- t.test(dados_capacitacao$Depois, 
                      dados_capacitacao$Antes,
                      paired = TRUE,
                      alternative = "two.sided",
                      conf.level = 0.95)

cat("Intervalo de Confiança 95% para μ_d:\n")
cat("[", round(ic_bilateral$conf.int[1], 3), " ; ", 
    round(ic_bilateral$conf.int[2], 3), "]\n\n")

cat("INTERPRETAÇÃO:\n")
cat("Com 95% de confiança, a melhora média verdadeira na pontuação\n")
cat("está entre", round(ic_bilateral$conf.int[1], 2), "e", 
    round(ic_bilateral$conf.int[2], 2), "pontos.\n\n")

# Diferentes níveis de confiança
niveis_conf <- c(0.90, 0.95, 0.99)
cat("INTERVALOS DE CONFIANÇA EM DIFERENTES NÍVEIS:\n")
cat("============================================\n")

resultados_ic <- data.frame(
  Nível = paste0(niveis_conf * 100, "%"),
  Inferior = NA,
  Superior = NA,
  Amplitude = NA
)

for(i in 1:length(niveis_conf)) {
  ic_temp <- t.test(dados_capacitacao$Depois, 
                   dados_capacitacao$Antes,
                   paired = TRUE,
                   conf.level = niveis_conf[i])
  
  resultados_ic$Inferior[i] <- round(ic_temp$conf.int[1], 3)
  resultados_ic$Superior[i] <- round(ic_temp$conf.int[2], 3)
  resultados_ic$Amplitude[i] <- round(ic_temp$conf.int[2] - ic_temp$conf.int[1], 3)
}

kable(resultados_ic, caption = "Intervalos de confiança para μ_d") %>%
  kable_styling(bootstrap_options = c("striped", "hover"))
```

16.7 Verificação de Pressupostos

Code
```{r}
#| label: pressupostos-emparelhados

cat("VERIFICAÇÃO DE PRESSUPOSTOS\n")
cat("===========================\n\n")

# 1. Teste de normalidade das diferenças
shapiro_test <- shapiro.test(diferenças)

cat("1. NORMALIDADE DAS DIFERENÇAS:\n")
cat("   Teste de Shapiro-Wilk:\n")
cat("   W =", round(shapiro_test$statistic, 4), "\n")
cat("   p-valor =", round(shapiro_test$p.value, 4), "\n")
cat("   Conclusão:", 
    ifelse(shapiro_test$p.value > 0.05, 
           "Não rejeitamos normalidade (p > 0.05)",
           "Rejeitamos normalidade (p ≤ 0.05)"), "\n\n")

# 2. Teste de outliers
Q1 <- quantile(diferenças, 0.25)
Q3 <- quantile(diferenças, 0.75)
IQR_val <- Q3 - Q1
limite_inf <- Q1 - 1.5 * IQR_val
limite_sup <- Q3 + 1.5 * IQR_val

outliers <- diferenças[diferenças < limite_inf | diferenças > limite_sup]

cat("2. DETECÇÃO DE OUTLIERS:\n")
cat("   Método: Regra 1.5×IQR\n")
cat("   Limite inferior:", round(limite_inf, 2), "\n")
cat("   Limite superior:", round(limite_sup, 2), "\n")
cat("   Outliers detectados:", length(outliers), "\n")
if(length(outliers) > 0) {
  cat("   Valores:", paste(outliers, collapse = ", "), "\n")
}
cat("\n")

# 3. Independência (teórica - depende do delineamento)
cat("3. INDEPENDÊNCIA:\n")
cat("   Este pressuposto deve ser avaliado pelo delineamento experimental.\n")
cat("   No exemplo: cada promotor é uma unidade independente.\n\n")

# 4. Resumo dos pressupostos
cat("RESUMO DOS PRESSUPOSTOS:\n")
cat("========================\n")
cat("✓ Dados emparelhados: SIM (mesmo promotor antes/depois)\n")
cat("✓ Normalidade:", 
    ifelse(shapiro_test$p.value > 0.05, "ATENDIDA", "QUESTIONÁVEL"), "\n")
cat("✓ Independência: ASSUMIDA (pelo delineamento)\n")
cat("✓ Outliers:", ifelse(length(outliers) == 0, "NENHUM", "PRESENTES"), "\n")
```

16.8 Exemplo 20.5: Comparação de Métodos de Ensino

Code
```{r}
#| label: exemplo-metodos-ensino

cat("EXEMPLO 20.5: COMPARAÇÃO DE MÉTODOS DE ENSINO\n")
cat("=============================================\n\n")

# Dados simulados baseados em cenário realista
# Estudantes de direito avaliados com dois métodos diferentes
# (cada estudante experimenta ambos os métodos)

set.seed(2024)
n_estudantes <- 15

# Simulando dados com correlação positiva entre métodos
notas_tradicional <- round(rnorm(n_estudantes, mean = 7.2, sd = 1.1), 1)
# Método ativo tem correlação com tradicional + efeito positivo
notas_ativo <- round(notas_tradicional + rnorm(n_estudantes, mean = 0.6, sd = 0.8), 1)
notas_ativo <- pmax(0, pmin(10, notas_ativo))  # Limitando entre 0-10

dados_ensino <- data.frame(
  Estudante = paste("Est", sprintf("%02d", 1:n_estudantes)),
  Método_Tradicional = notas_tradicional,
  Método_Ativo = notas_ativo
) %>%
  mutate(Diferença = Método_Ativo - Método_Tradicional)

# Exibindo os dados

  kable_styling(bootstrap_options = c("striped", "hover"))

# Teste bilateral (pode haver diferença em qualquer direção)
teste_ensino <- t.test(dados_ensino$Método_Ativo, 
                      dados_ensino$Método_Tradicional,
                      paired = TRUE,
                      alternative = "two.sided",
                      conf.level = 0.95)

cat("\nTESTE DE HIPÓTESES:\n")
cat("H₀: μ_d = 0 (métodos são equivalentes)\n")
cat("H₁: μ_d ≠ 0 (métodos diferem)\n")
cat("α = 0.05\n\n")

cat("RESULTADOS:\n")
cat("t =", round(teste_ensino$statistic, 4), "\n")
cat("gl =", teste_ensino$parameter, "\n")
cat("p-valor =", round(teste_ensino$p.value, 4), "\n")
cat("IC 95%: [", round(teste_ensino$conf.int[1], 3), " ; ", 
    round(teste_ensino$conf.int[2], 3), "]\n\n")

cat("CONCLUSÃO:", 
    ifelse(teste_ensino$p.value < 0.05,
```

16.9 Tamanho do Efeito e Poder do Teste

Code
```{r}
#| label: tamanho-efeito-poder

# Calculando o tamanho do efeito (d de Cohen para amostras pareadas)
d_cohen <- mean(diferenças) / sd(diferenças)

cat("TAMANHO DO EFEITO\n")
cat("=================\n")
cat("d de Cohen =", round(d_cohen, 3), "\n")
cat("Interpretação:", 
    ifelse(abs(d_cohen) < 0.2, "Efeito pequeno",
           ifelse(abs(d_cohen) < 0.5, "Efeito pequeno a médio",
                  ifelse(abs(d_cohen) < 0.8, "Efeito médio",
                         "Efeito grande"))), "\n\n")

# Análise de poder (cálculo aproximado)
cat("ANÁLISE DE PODER (aproximada):\n")
cat("==============================\n")

# Calculando poder para diferentes tamanhos amostrais
tamanhos_n <- c(5, 10, 15, 20, 30, 50)
niveis_alfa <- c(0.01, 0.05, 0.10)

poder_matrix <- matrix(NA, nrow = length(tamanhos_n), ncol = length(niveis_alfa))
rownames(poder_matrix) <- paste("n =", tamanhos_n)
colnames(poder_matrix) <- paste("α =", niveis_alfa)

# Simulação aproximada do poder
for(i in 1:length(tamanhos_n)) {
  for(j in 1:length(niveis_alfa)) {
    # Valor não-centrality parameter
    ncp <- abs(d_cohen) * sqrt(tamanhos_n[i])
    # Valor crítico
    t_crit <- qt(1 - niveis_alfa[j]/2, df = tamanhos_n[i] - 1)
    # Poder aproximado
    poder_matrix[i, j] <- round(1 - pt(t_crit, df = tamanhos_n[i] - 1, ncp = ncp) + 
                               pt(-t_crit, df = tamanhos_n[i] - 1, ncp = ncp), 3)
  }
}

cat("TABELA DE PODER (para d =", round(d_cohen, 3), "):\n")
print(poder_matrix)
```
